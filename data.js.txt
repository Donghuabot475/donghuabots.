// Utility functions
function getDonghuaById(id) {
    return DONGHUA_DATA.find(d => d.id === id);

function getFeaturedDonghua() {
    return DONGHUA_DATA.filter(d => d.featured);
}

function getTrendingDonghua() {
    return DONGHUA_DATA.filter(d => d.trending).slice(0, 6);
}

function getPopularDonghua() {
    return [...DONGHUA_DATA].sort((a, b) => a.popular - b.popular).slice(0, 10);
}

function getLatestEpisodes() {
    return [...DONGHUA_DATA]
        .sort((a, b) => b.currentEpisode - a.currentEpisode)
        .slice(0, 8)
        .map(d => ({
            ...d,
            displayEpisode: d.currentEpisode,
            isNew: d.currentEpisode === d.totalEpisodes
        }));
}

function getDonghuaByGenre(genre) {
    if (!genre || genre === "All") return DONGHUA_DATA;
    return DONGHUA_DATA.filter(d => d.genres.includes(genre));
}

function searchDonghua(query) {
    if (!query) return DONGHUA_DATA;
    const lowerQuery = query.toLowerCase();
    return DONGHUA_DATA.filter(d =>
        d.title.toLowerCase().includes(lowerQuery) ||
        d.altTitle.toLowerCase().includes(lowerQuery) ||
        d.genres.some(g => g.toLowerCase().includes(lowerQuery)) ||
        d.description.toLowerCase().includes(lowerQuery) ||
        d.currentEpisode.toString().includes(lowerQuery)
    );
}

function getRecommendations(excludeId) {
